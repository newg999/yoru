# --- Sistema base (lo que una Fedora mínima no trae) -------------------------
mesa-dri-drivers          # gráficos (OpenGL)
mesa-vulkan-drivers       # gráficos (Vulkan)
pipewire                  # sonido
pipewire-pulseaudio       # compatibilidad con apps de PulseAudio (y la barra)
pipewire-alsa
wireplumber               # gestor de sesión de audio (wpctl)
pipewire-utils            # pw-dump: el menú de audio lo usa para ver las salidas
pulseaudio-utils          # pactl: el menú de audio lo usa para cambiar de salida
alsa-sof-firmware         # sonido en portátiles Intel modernos (sin él, mudos)
alsa-ucm                  # perfiles de las tarjetas de sonido (van con el anterior)
NetworkManager-wifi       # wifi
NetworkManager-tui        # nmtui: arreglar el wifi desde la consola si algo falla
bluez                     # bluetooth
upower                    # estado de la batería
tuned-ppd                 # perfiles de energía (ahorro / equilibrado / rendimiento)
xdg-user-dirs             # crea Descargas, Imágenes, Documentos...
xdg-utils                 # xdg-open: abrir archivos y enlaces con su app
flatpak                   # apps de Flathub (la tienda: Mod+Alt+S)
dconf                     # guarda los ajustes de las apps GTK (gsettings)
gvfs                      # papelera, unidades USB y red en el gestor de archivos
gvfs-mtp                  # móviles Android por USB
gvfs-smb                  # carpetas compartidas de Windows/Samba
google-noto-sans-fonts    # fuente general de las apps
google-noto-color-emoji-fonts  # emojis 🎉

# --- Inicio de sesión --------------------------------------------------------
# Solo se activa si no tienes ya uno (por ejemplo, GDM de GNOME)
greetd                    # gestor de inicio de sesión
gtkgreet                  # pantalla de login gráfica (system/greetd/)
jetbrains-mono-fonts      # su fuente (la Nerd Font va solo en tu usuario)

# --- Núcleo de Niri ----------------------------------------------------------
niri                      # el compositor
xwayland-satellite        # apps X11 (Steam, Discord...). Niri lo arranca solo
xdg-desktop-portal-gnome  # compartir pantalla, capturas
xdg-desktop-portal-gtk    # selector de archivos y ajustes para apps
gnome-keyring             # guarda contraseñas (wifi, navegador...)
gnome-keyring-pam         # lo desbloquea al iniciar sesión (si no, Brave pide contraseña)
qt6-qtwayland             # apps Qt (VLC, Telegram...) nativas en Wayland
qt5-qtwayland
libva-intel-media-driver  # aceleración de vídeo en gráficas Intel (menos CPU y batería)
mate-polkit               # ventana que pide la contraseña de administrador

# --- Escritorio --------------------------------------------------------------
waybar                    # barra superior
rofi                      # lanzador de apps y menús (wifi, bluetooth...)
ImageMagick               # miniaturas del selector de fondos
papirus-icon-theme        # iconos
adw-gtk3-theme            # tema oscuro para apps GTK3 (LibreOffice, ajustes de red...)
grim                      # hace la captura (la usa Mod+Shift+S)
slurp                     # elegir una zona de la pantalla con el ratón
swappy                    # editor de capturas (flechas, texto, difuminar)
mako                      # notificaciones
swaybg                    # fondo de pantalla
swaylock                  # pantalla de bloqueo
swayidle                  # bloqueo automático por inactividad
kitty                     # terminal

brave-browser             # navegador (install.sh añade antes su repositorio)
nautilus                  # gestor de archivos (Mod+E)
file-roller               # comprimir/descomprimir (integrado en Nautilus)
loupe                     # visor de imágenes
mpv                       # vídeo
okular                    # visor de PDF (colores oscuros: config/kdeglobals)

pavucontrol               # mezclador de audio
blueman                   # bluetooth
network-manager-applet    # icono de red y editor de conexiones

brightnessctl             # teclas de brillo

libnotify                 # notify-send
htop                      # monitor del sistema (clic en la CPU de la barra)
btop                      # monitor del sistema más completo y bonito

wget                      # descargar archivos desde la terminal
fontconfig                # fc-cache
